var g_kalman = "Kalman Filter Settings"

i_kalmanMomentum = input.float(0.3, "Momentum", minval=0.1, maxval=0.9, step=0.05, group=g_kalman)

// Supertrend Settings
var g_st = "Supertrend Settings"
i_atrPeriod = input.int(10, "ATR Period", minval=1, group=g_st)
i_atrMultiplier = input.float(3.0, "ATR Multiplier", minval=0.1, step=0.1, group=g_st)

// VWMA Bands Settings
var g_vwma = "VWMA Bands (OB/OS Zones)"
i_vwmaLength = input.int(20, "VWMA Length", minval=1, group=g_vwma)
i_obMultiplier = input.float(1.5, "Overbought Multiplier", minval=0.5, step=0.1, group=g_vwma)
i_osMultiplier = input.float(1.5, "Oversold Multiplier", minval=0.5, step=0.1, group=g_vwma)
i_bandLookback = input.int(20, "Band Lookback", minval=5, group=g_vwma)

// Take Profit Settings

i_enableTP = input.bool(true, "Enable Multiple TP Levels", group=g_tp)
i_tp1RR = input.float(0.5, "TP1 Risk:Reward", minval=0.3, step=0.1, group=g_tp)
i_tp2RR = input.float(1.0, "TP2 Risk:Reward", minval=0.5, step=0.1, group=g_tp)
i_tp3RR = input.float(1.5, "TP3 Risk:Reward", minval=0.8, step=0.1, group=g_tp)

i_enableBreakout = input.bool(true, "Enable Breakout Signals", group=g_breakout)

i_enableOrderBlocks = input.bool(true, "Enable Order Blocks", group=g_ob)
i_maxOrderBlocks = input.int(10, "Max Order Blocks on Chart", minval=5, maxval=50, group=g_ob)

// FUNCTIONS
// ══════════════════════════════════════════════════════════════════════════════════════════════════════════════════════

f_kalmanFilter(float src, float gain, float momentum) =>
    var float estimate = na
    var float velocity = 0.0
    
    if na(estimate)
        estimate := src
        velocity := 0.0
    else
        float prediction = estimate + velocity
        float error = src - prediction
        estimate := prediction + gain * error
        velocity := velocity * momentum + gain * error
    
    estimate

// ══════════════════════════════════════════════════════════════════════════════════════════════════════════════════════
// CALCULATIONS
// ══════════════════════════════════════════════════════════════════════════════════════════════════════════════════════

// --- Kalman Filtering ---
float kalmanHL2 = f_kalmanFilter((high + low) / 2, i_kalmanGain, i_kalmanMomentum)

// --- ATR Calculation ---
float atrValue = ta.rma(ta.tr(true), i_atrPeriod)
float atrFiltered = f_kalmanFilter(atrValue, i_kalmanGain * 0.5, i_kalmanMomentum)

// --- Supertrend Bands ---
float upperBand = kalmanHL2 + i_atrMultiplier * atrFiltered
float lowerBand = kalmanHL2 - i_atrMultiplier * atrFiltered

// --- Final Bands (Trailing) ---
var float finalUpperBand = na
var float finalLowerBand = na

finalUpperBand := na(finalUpperBand[1]) or close[1] > finalUpperBand[1] ? upperBand : math.min(upperBand, finalUpperBand[1])
finalLowerBand := na(finalLowerBand[1]) or close[1] < finalLowerBand[1] ? lowerBand : math.max(lowerBand, finalLowerBand[1])

// --- Trend Detection ---
var int trend = 1
int prevTrend = trend
trend := close > finalUpperBand[1] ? 1 : close < finalLowerBand[1] ? -1 : nz(trend[1], 1)
float supertrend = trend == 1 ? finalLowerBand : finalUpperBand

// --- Primary Signal Generation ---
bool buySignal = trend == 1 and prevTrend == -1
bool sellSignal = trend == -1 and prevTrend == 1

// --- VWMA Bands ---
float vwma = ta.vwma(close, i_vwmaLength)
float hh = ta.highest(high, i_bandLookback)
float ll = ta.lowest(low, i_bandLookback)
float rangeHigh = hh - vwma
float rangeLow = vwma - ll
float obZone = vwma + rangeHigh * i_obMultiplier
float osZone = vwma - rangeLow * i_osMultiplier

// --- Overbought/Oversold Detection ---
bool isOB = high >= obZone
bool isOS = low <= osZone
bool wasOB = high[1] >= obZone[1]
bool wasOS = low[1] <= osZone[1]

// --- Reversal Signals ---
bool reversalDown = wasOB and close < obZone and close < open
bool reversalUp = wasOS and close > osZone and close > open

// --- Breakout Detection ---
float highestHigh = ta.highest(high, i_breakoutLookback)
float lowestLow = ta.lowest(low, i_breakoutLookback)

bool breakoutBuy = i_enableBreakout and trend == 1 and close > highestHigh[1] and (close - open) > atrFiltered * i_breakoutStrength and not buySignal
bool breakoutSell = i_enableBreakout and trend == -1 and close < lowestLow[1] and (open - close) > atrFiltered * i_breakoutStrength and not sellSignal

// --- Order Blocks Array Management ---
var array<box> allOrderBlocks = array.new<box>()
var array<int> obCreationBars = array.new<int>()

// Clean boxes that are too old (beyond extension)
if array.size(allOrderBlocks) > 0
    for i = array.size(allOrderBlocks) - 1 to 0
        if bar_index - array.get(obCreationBars, i) > i_obExtension + 50
            box.delete(array.get(allOrderBlocks, i))
            array.remove(allOrderBlocks, i)
            array.remove(obCreationBars, i)

// Clean if too many boxes
if array.size(allOrderBlocks) > i_maxOrderBlocks
    box.delete(array.shift(allOrderBlocks))
    array.shift(obCreationBars)

// Detect and Draw Order Blocks
if i_enableOrderBlocks and buySignal
    // Find last bearish candle before buy signal
    for i = 1 to 10
        if close[i] < open[i] and (high[i] - low[i]) > atrFiltered * 0.3
            newBox = box.new(bar_index - i, high[i], bar_index + i_obExtension, low[i],
                           border_color=color.new(color.green, 0),
                           bgcolor=color.new(color.green, 85),
                           border_width=2,
                           extend=extend.none)
            array.push(allOrderBlocks, newBox)
            array.push(obCreationBars, bar_index)
            break

if i_enableOrderBlocks and sellSignal
    // Find last bullish candle before sell signal
    for i = 1 to 10
        if close[i] > open[i] and (high[i] - low[i]) > atrFiltered * 0.3
            newBox = box.new(bar_index - i, high[i], bar_index + i_obExtension, low[i],
                           border_color=color.new(color.red, 0),
                           bgcolor=color.new(color.red, 85),
                           border_width=2,
                           extend=extend.none)
            array.push(allOrderBlocks, newBox)
            array.push(obCreationBars, bar_index)
            break

// --- TP Levels Management ---
var float entryPrice = na
var float stopLoss = na
var int activeSignal = 0
var int entryBar = 0

if buySignal
    entryPrice := close
    stopLoss := finalLowerBand
    activeSignal := 1
    entryBar := bar_index

if sellSignal
    entryPrice := close
    stopLoss := finalUpperBand
    activeSignal := -1
    entryBar := bar_index

// Reset if trend changes
if trend != activeSignal
    activeSignal := 0
    entryPrice := na
    stopLoss := na

// Calculate TP levels
var float tp1Level = na
var float tp2Level = na
var float tp3Level = na
var bool tp1Hit = false
var bool tp2Hit = false
var bool tp3Hit = false

if activeSignal == 1 and not na(entryPrice) and not na(stopLoss)
    float riskDistance = math.abs(entryPrice - stopLoss)
    tp1Level := entryPrice + riskDistance * i_tp1RR
    tp2Level := entryPrice + riskDistance * i_tp2RR
    tp3Level := entryPrice + riskDistance * i_tp3RR
    
    if not tp1Hit and high >= tp1Level
        tp1Hit := true
    if not tp2Hit and high >= tp2Level
        tp2Hit := true
    if not tp3Hit and high >= tp3Level
        tp3Hit := true

if activeSignal == -1 and not na(entryPrice) and not na(stopLoss)
    float riskDistance = math.abs(stopLoss - entryPrice)
    tp1Level := entryPrice - riskDistance * i_tp1RR
    tp2Level := entryPrice - riskDistance * i_tp2RR
    tp3Level := entryPrice - riskDistance * i_tp3RR
    
    if not tp1Hit and low <= tp1Level
        tp1Hit := true
    if not tp2Hit and low <= tp2Level
        tp2Hit := true
    if not tp3Hit and low <= tp3Level
        tp3Hit := true

if buySignal or sellSignal
    tp1Hit := false
    tp2Hit := false
    tp3Hit := false

// @function Kalman Filter - Reduces noise and smooths price data
// @param src Source price series
// @param gain Kalman gain parameter
// @param momentum Momentum parameter
// @returns Filtered price value
f_kalmanFilter(float src, float gain, float momentum) =>
    var float estimate = na
    var float velocity = 0.0
    
    if na(estimate)
        estimate := src
        velocity := 0.0
    else
        float prediction = estimate + velocity
        float error = src - prediction
        estimate := prediction + gain * error
        velocity := velocity * momentum + gain * error
    
    estimate

// ══════════════════════════════════════════════════════════════════════════════════════════════════════════════════════
// CALCULATIONS
// ══════════════════════════════════════════════════════════════════════════════════════════════════════════════════════

// --- Kalman Filtering ---
float kalmanHL2 = f_kalmanFilter((high + low) / 2, i_kalmanGain, i_kalmanMomentum)

// --- ATR Calculation ---
float atrValue = ta.rma(ta.tr(true), i_atrPeriod)
float atrFiltered = f_kalmanFilter(atrValue, i_kalmanGain * 0.5, i_kalmanMomentum)

// --- Supertrend Bands ---
float upperBand = kalmanHL2 + i_atrMultiplier * atrFiltered
float lowerBand = kalmanHL2 - i_atrMultiplier * atrFiltered

// --- Final Bands (Trailing) ---
var float finalUpperBand = na
var float finalLowerBand = na

finalUpperBand := na(finalUpperBand[1]) or close[1] > finalUpperBand[1] ? upperBand : math.min(upperBand, finalUpperBand[1])
finalLowerBand := na(finalLowerBand[1]) or close[1] < finalLowerBand[1] ? lowerBand : math.max(lowerBand, finalLowerBand[1])

// --- Trend Detection ---
var int trend = 1
int prevTrend = trend
trend := close > finalUpperBand[1] ? 1 : close < finalLowerBand[1] ? -1 : nz(trend[1], 1)
float supertrend = trend == 1 ? finalLowerBand : finalUpperBand

// --- Signal Generation ---
bool buySignal = trend == 1 and prevTrend == -1
bool sellSignal = trend == -1 and prevTrend == 1

// --- Store Last Entry ---
var float entryPrice = na
var string entryType = ""

if buySignal
    entryPrice := close
    entryType := "BUY"

if sellSignal
    entryPrice := close
    entryType := "SELL"

// --- VWMA Bands (Overbought/Oversold Zones) ---
float vwma = ta.vwma(close, i_vwmaLength)
float hh = ta.highest(high, i_bandLookback)
float ll = ta.lowest(low, i_bandLookback)
float rangeHigh = hh - vwma
float rangeLow = vwma - ll
float obZone = vwma + rangeHigh * i_obMultiplier
float osZone = vwma - rangeLow * i_osMultiplier

// --- Overbought/Oversold Detection ---
bool isOB = high >= obZone
bool isOS = low <= osZone
bool wasOB = high[1] >= obZone[1]
bool wasOS = low[1] <= osZone[1]

// --- Reversal Signals ---
bool reversalDown = wasOB and close < obZone and close < open
bool reversalUp = wasOS and close > osZone and close > open

// --- Take Profit Logic ---
var float tpLevelBuy = na
var float tpLevelSell = na
var bool tpBuyHit = false
var bool tpSellHit = false

// Set TP levels on new signals
if buySignal
    tpLevelBuy := obZone
    tpBuyHit := false

if sellSignal
    tpLevelSell := osZone
    tpSellHit := false

// Check TP conditions
bool higherHigh = high > ta.highest(high[1], 2)
bool lowerLow = low < ta.lowest(low[1], 2)
bool tpBuy = false
bool tpSell = false

if trend == 1 and not tpBuyHit and not na(tpLevelBuy)
    if high >= tpLevelBuy or (isOB and higherHigh)
        tpBuy := true
        tpBuyHit := true

if trend == -1 and not tpSellHit and not na(tpLevelSell)
    if low <= tpLevelSell or (isOS and lowerLow)
        tpSell := true
        tpSellHit := true
